# 电机测试
```python
from machine import Pin, PWM
import time

# --- 准确的 GPIO 映射 ---
PIN_IN1_A = 13  # Motor A IN1
PIN_IN2_A = 15  # Motor A IN2
PIN_IN1_B = 25  # Motor B IN1
PIN_IN2_B = 14  # Motor B IN2

# --- 电机 PWM 初始化 ---
PWM_FREQ = 1000 # 1kHz 

motor_a_in1 = PWM(Pin(PIN_IN1_A, Pin.OUT), freq=PWM_FREQ)
motor_a_in2 = PWM(Pin(PIN_IN2_A, Pin.OUT), freq=PWM_FREQ)

motor_b_in1 = PWM(Pin(PIN_IN1_B, Pin.OUT), freq=PWM_FREQ)
motor_b_in2 = PWM(Pin(PIN_IN2_B, Pin.OUT), freq=PWM_FREQ)

def stop_motors():
    """强制停止所有电机"""
    motor_a_in1.duty(0)
    motor_a_in2.duty(0)
    motor_b_in1.duty(0)
    motor_b_in2.duty(0)
    print("Motors Stopped")

def test_motor_a(speed, duration):
    """
    测试电机 A
    speed 范围: -1023 到 1023。正数正转，负数反转。
    """
    print(f"Testing Motor A at speed {speed}")
    if speed > 0:
        motor_a_in1.duty(speed)
        motor_a_in2.duty(0)
    elif speed < 0:
        motor_a_in1.duty(0)
        motor_a_in2.duty(abs(speed))
    else:
        stop_motors()
        
    time.sleep(duration)
    stop_motors()

def test_motor_b(speed, duration):
    """
    测试电机 B
    speed 范围: -1023 到 1023。正数正转，负数反转。
    """
    print(f"Testing Motor B at speed {speed}")
    if speed > 0:
        motor_b_in1.duty(speed)
        motor_b_in2.duty(0)
    elif speed < 0:
        motor_b_in1.duty(0)
        motor_b_in2.duty(abs(speed))
    else:
        stop_motors()
        
    time.sleep(duration)
    stop_motors()

# --- 测试流程 ---
try:
    stop_motors() 
    time.sleep(1)

    # 初始占空比设为 600
    test_speed = 600 

    # 测试 A 电机
    test_motor_a(test_speed, 2)
    time.sleep(0.5)
    test_motor_a(-test_speed, 2)
    time.sleep(1)

    # 测试 B 电机
    test_motor_b(test_speed, 2)
    time.sleep(0.5)
    test_motor_b(-test_speed, 2)
    
    print("Test complete.")

except KeyboardInterrupt:
    stop_motors()
    print("Test interrupted by user.")
```
# 传感器测试
```python
from machine import Pin
import time

# --- 五路循迹引脚定义（从左到右：ADC_1 到 ADC_5） ---
TRACKING_PINS = [27, 33, 32, 35, 34]

# 初始化引脚为输入模式
# 注意：GPIO 34/35 没有内部上拉功能，这里统一用 Pin.IN 即可
sensors = [Pin(pin, Pin.IN) for pin in TRACKING_PINS]

# 每个传感器对应的权重（用于计算偏差）
# 最左边偏得最远设为 -2，中间为 0，最右边设为 2
WEIGHTS = [-2, -1, 0, 1, 2]

def read_tracking_sensors():
    """
    读取五路循迹的状态
    假设：检测到黑线输出 1，检测到白底输出 0 
    (如果你的传感器逻辑反了，可以在下面取反，即 1 - s.value())
    """
    states = []
    for s in sensors:
        states.append(s.value())
    return states

def calculate_error(states):
    """
    根据五路状态，计算小车偏离黑线的中心偏差值 (Error)
    返回范围：-2.0 (极度偏左) 到 +2.0 (极度偏右)，0.0 表示正对中央
    """
    total_weight = 0
    active_count = 0
    
    for i in range(5):
        if states[i] == 1: # 假设 1 代表压到黑线
            total_weight += WEIGHTS[i] # 根据权重计算总偏差
            active_count += 1 # 压线的传感器数量
            
    if active_count == 0:
        # 没有任何传感器压线（可能冲出跑道了），返回 None 特殊处理
        return None
        
    return total_weight / active_count # 为什么这么算？因为如果多个传感器同时压线，我们取平均偏差值，这样更平滑一些。

# --- 主测试循环 ---
print("五路循迹传感器测试开始...")
time.sleep(1)

while True:
    try:
        # 1. 读取原始数据
        raw_states = read_tracking_sensors()
        
        # 2. 计算偏差值
        error = calculate_error(raw_states)
        
        # 3. 格式化打印输出，方便在串口监视器里直观查看
        # 打印类似：[0, 0, 1, 0, 0]  偏差值: 0.00
        state_str = ", ".join(str(x) for x in raw_states)
        if error is not None:
            print(f"传感器状态: [{state_str}] -> 偏差值: {error:+.2f}")
        else:
            print(f"传感器状态: [{state_str}] -> 丢线状态！")
            
        time.sleep(0.2) # 每 200ms 刷新一次
        
    except KeyboardInterrupt:
        print("\n测试结束")
        break
```
